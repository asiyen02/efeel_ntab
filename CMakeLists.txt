cmake_minimum_required(VERSION 3.24)
option(HOST_TEST "Build portable driver tests instead of MCU firmware" OFF)
if(NOT HOST_TEST)
  set(CMAKE_SYSTEM_NAME Generic)
  set(CMAKE_SYSTEM_PROCESSOR arm)
  set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
  find_program(ARM_GCC arm-none-eabi-gcc REQUIRED)
  set(CMAKE_C_COMPILER "${ARM_GCC}")
  set(CMAKE_ASM_COMPILER "${ARM_GCC}")
endif()
project(ultratac C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(APP_WARNINGS -Wall -Wextra -Wshadow -Wconversion -Werror)
if(HOST_TEST)
  enable_testing()
  add_executable(test_drive drive.c test_drive.c)
  target_compile_options(test_drive PRIVATE ${APP_WARNINGS})
  add_test(NAME driver COMMAND test_drive)
  return()
endif()
# Cached source directories may be supplied for offline builds. Otherwise fetch pinned archives.
include(FetchContent)
function(dependency name repo revision sha)
  if(NOT ${name}_DIR)
    FetchContent_Declare(${name} URL "https://api.github.com/repos/${repo}/zipball/${revision}" URL_HASH "SHA256=${sha}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(${name})
    string(TOLOWER "${name}" lower_name)
    set(${name}_DIR "${${lower_name}_SOURCE_DIR}" PARENT_SCOPE)
  endif()
endfunction()
